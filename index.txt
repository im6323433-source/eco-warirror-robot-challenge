<script>
const benefitsBtn=document.getElementById('benefitsBtn');


function goTo(id){
  document.querySelectorAll('.page').forEach(p=>p.classList.remove('active'));
  document.getElementById(id).classList.add('active');
  window.scrollTo(0,0);
  if(id==='page-benefits'){benefitsBtn.style.display='none';}
}


/* Start Game Button */
function startGameAndGo(){
  goTo('page-game');
  startGame();
}


/* Game Logic */
const gameArea=document.getElementById('gameArea'),
      robot=document.getElementById('robot'),
      scoreBoard=document.getElementById('scoreBoard'),
      gameTimerEl=document.getElementById('gameTimer'),
      quizBtn=document.getElementById('quizBtn');


let gameWidth,gameHeight;
function resizeGame(){gameWidth=gameArea.clientWidth;gameHeight=gameArea.clientHeight;}
window.addEventListener('resize',resizeGame);resizeGame();


let robotPos={x:10,y:10},items=[],score=0,gameTime=30,gameInterval;
const envEmojis=["🌳","♻️","💧","🗑️","🌱","🍃","☀️"];


function placeItems(count=10){
    items.forEach(it=>it.remove()); items=[];
    for(let i=0;i<count;i++){
        const it=document.createElement('div');
        it.className='item';
        it.textContent=envEmojis[Math.floor(Math.random()*envEmojis.length)];
        it.style.left=Math.floor(Math.random()*(gameWidth-60))+'px';
        it.style.top=Math.floor(Math.random()*(gameHeight-60))+'px';
        gameArea.appendChild(it);
        items.push(it);
    }
}
robot.style.left=robotPos.x+'px';
robot.style.top=robotPos.y+'px';
placeItems(10);


function checkCollision(){
    const rRect=robot.getBoundingClientRect();
    for(let i=items.length-1;i>=0;i--){
        const itRect=items[i].getBoundingClientRect();
        if(!(rRect.right<itRect.left||rRect.left>itRect.right||rRect.bottom<itRect.top||rRect.top>itRect.bottom)){
            items[i].remove();
            items.splice(i,1);
            score++;
            scoreBoard.textContent='Score: '+score;
        }
    }
}


document.addEventListener('keydown',e=>{
    const active=document.querySelector('.page.active');
    if(!active||active.id!=='page-game')return;
    const step=16;
    if(['ArrowUp','ArrowDown','ArrowLeft','ArrowRight'].includes(e.key))e.preventDefault();
    if(e.key==='ArrowUp')robotPos.y=Math.max(0,robotPos.y-step);
    if(e.key==='ArrowDown')robotPos.y=Math.min(gameHeight-robot.clientHeight,robotPos.y+step);
    if(e.key==='ArrowLeft')robotPos.x=Math.max(0,robotPos.x-step);
    if(e.key==='ArrowRight')robotPos.x=Math.min(gameWidth-robot.clientWidth,robotPos.x+step);
    robot.style.left=robotPos.x+'px';
    robot.style.top=robotPos.y+'px';
    checkCollision();
});


function startGame(){
    score=0; robotPos={x:10,y:10};
    robot.style.left=robotPos.x+'px';
    robot.style.top=robotPos.y+'px';
    placeItems(10);
    gameTime=30;
    gameTimerEl.textContent='Time left: 30s';
    quizBtn.style.display='inline-block';
    clearInterval(gameInterval);
    gameInterval=setInterval(()=>{
        gameTime--;
        gameTimerEl.textContent='Time left: '+gameTime+'s';
        if(gameTime<=0){clearInterval(gameInterval);}
    },1000);
}


/* Quiz Logic */
const quizQuestions=[
    {q:"1) What is the most recyclable material?",options:["Plastic","Glass","Metal","Paper"],ans:1},
    {q:"2) Which gas is mainly responsible for global warming?",options:["Oxygen","Nitrogen","Carbon Dioxide","Hydrogen"],ans:2},
    {q:"3) Which activity helps conserve water?",options:["Leaving taps open","Fixing leaks","Overwatering plants","Washing car daily"],ans:1},
    {q:"4) What should you do with electronic waste?",options:["Throw in trash","Recycle properly","Burn it","Bury it"],ans:1},
    {q:"5) Planting trees helps to:",options:["Pollute air","Reduce CO2","Increase soil erosion","Waste water"],ans:1},
    {q:"6) What is composting?",options:["Burning waste","Recycling food waste","Throwing in river","Using chemicals"],ans:1},
    {q:"7) Energy saving can be done by:",options:["Leaving lights on","Using LED bulbs","Overcharging devices","Wasting electricity"],ans:1}
];


let quizIndex=0,correctCount=0,wrongCount=0;
const quizArea=document.getElementById('quizArea'),quizAnswer=document.getElementById('quizAnswer');


function startQuiz(){
    quizIndex=0; correctCount=0; wrongCount=0;
    renderQuiz();
}


function renderQuiz(){
    quizArea.innerHTML=''; quizAnswer.textContent='';
    if(quizIndex>=quizQuestions.length){
        quizArea.innerHTML=`<div class="quiz-question">🎉 Quiz Completed!</div>
                            <p>Correct Answers: ${correctCount}</p>
                            <p>Wrong Answers: ${wrongCount}</p>`;
        benefitsBtn.style.display='inline-block';
        return;
    }
    const q=quizQuestions[quizIndex];
    const qDiv=document.createElement('div'); qDiv.className='quiz-question'; qDiv.textContent=q.q;
    quizArea.appendChild(qDiv);


    q.options.forEach((opt,i)=>{
        const btn=document.createElement('div');
        btn.className='quiz-option';
        btn.textContent=opt;
        btn.tabIndex=0;
        btn.addEventListener('click',()=>selectOption(i,q.ans,btn));
        btn.addEventListener('keydown',e=>{if(e.key==='Enter'||e.keyCode===13){e.preventDefault();selectOption(i,q.ans,btn);}});
        quizArea.appendChild(btn);
    });
}


function selectOption(sel,correct,btn){
    document.querySelectorAll('.quiz-option').forEach(b=>b.style.pointerEvents='none');
    if(sel===correct){btn.classList.add('correct'); quizAnswer.textContent='✅ Correct!'; correctCount++;}
    else{btn.classList.add('wrong'); quizAnswer.textContent=`❌ Wrong! Correct: ${quizQuestions[quizIndex].options[correct]}`; wrongCount++;}
    setTimeout(()=>{quizIndex++; renderQuiz();},1500);
}
